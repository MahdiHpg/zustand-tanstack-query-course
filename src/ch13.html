پروژهٔ عملی ۲ — اپ کامل با Zustand و TanStack Query

اپ «پست‌بلاگ» را از صفر می‌سازیم: ورود کاربر، جستجو و صفحه‌بندی، انتشار/ویرایش/حذف خوش‌بینانه — با جداسازی تمیز UI State و Server State در یک پروژهٔ واحد.

در این فصل یاد می‌گیرید

  • تقسیم کار نهایی بین Zustand و TanStack Query در یک پروژهٔ واقعی
  • اتصال auth (Zustand) به لایهٔ API و به کش (removal در خروج)
  • الگوی «فیلتر در استور → کلید در Query» برای جستجو و صفحه‌بندی
  • ترکیب همهٔ الگوها: mutation، invalidation، optimistic و prefetch

«فیلترها در Zustand زندگی می‌کنند، ولی داده فقط از دروازهٔ کلیدِ Query می‌گذرد»: فیلتر جستجو State زندهٔ UI است، اما هر ترکیب فیلتر = یک کلید کوئری متفاوت = یک ورودی کش مستقل. هیچ دادهٔ سرور به استور فیلترها وارد نمی‌شود.

۱) لایهٔ API + Auth Store

src/stores/useAuthStore.jsjs
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

export const useAuthStore = create(
  persist(
    (set) => ({
      token: null,
      user: null,
      setSession: ({ token, user }) => set({ token, user }),
      clear: () => set({ token: null, user: null }),
    }),
    { name: 'auth', partialize: (s) => ({ token: s.token, user: s.user }) }
  )
)
src/lib/api.jsjs
import { useAuthStore } from '../stores/useAuthStore'
const BASE = 'https://jsonplaceholder.typicode.com'

async function request(path, { method = 'GET', body } = {}) {
  const headers = { 'Content-Type': 'application/json' }
  // بیرونِ React: خواندن توکن از استور، نه از هوک
  const token = useAuthStore.getState().token
  if (token) headers.Authorization = `Bearer ${token}`

  const res = await fetch(BASE + path, {
    method,
    headers,
    body: body ? JSON.stringify(body) : undefined,
  })
  if (res.status === 401) {
    useAuthStore.getState().clear() // نشست منقضی — یک‌جا و متمرکز
    throw new Error('نشست منقضی شده؛ دوباره وارد شوید')
  }
  if (!res.ok) throw new Error(`API error ${res.status}`)
  return res.status === 204 ? null : res.json()
}

export const api = {
  posts: (params = '', signal) => request(`/posts${params}`, { signal }),
  post: (id, signal) => request(`/posts/${id}`, { signal }),
  addPost: (post) => request('/posts', { method: 'POST', body: post }),
  updatePost: (id, patch) => request(`/posts/${id}`, { method: 'PATCH', body: patch }),
  deletePost: (id) => request(`/posts/${id}`, { method: 'DELETE' }),
  login: async (email) => {
    // JSONPlaceholder لاگین ندارد؛ شبیه‌سازی: کاربر را از /users می‌یابیم
    const users = await request(`/users?email=${encodeURIComponent(email)}`)
    if (!users.length) throw new Error('کاربری با این ایمیل نیست')
    return { user: users[0], token: `demo-token-${users[0].id}` }
  },
}

import { create } from 'zustand'

export const usePostFiltersStore = create((set) => ({
  q: '',
  page: 1,
  setQ: (q) => set({ q, page: 1 }),  // جستجوی جدید یعنی برگشت به صفحهٔ ۱
  nextPage: () => set((s) => ({ page: s.page + 1 })),
  prevPage: () => set((s) => ({ page: Math.max(1, s.page - 1) })),
}))
src/features/posts/keys.js و queries.jsjs
import { queryOptions, keepPreviousData } from '@tanstack/react-query'
import { api } from '../../lib/api'

export const postKeys = {
  all: ['posts'],
  lists: () => [...postKeys.all, 'list'],
  list: (filters) => [...postKeys.lists(), filters],
  details: () => [...postKeys.all, 'detail'],
  detail: (id) => [...postKeys.details(), id],
}

export const postQueries = {
  list: (filters) =>
    queryOptions({
      queryKey: postKeys.list(filters),
      queryFn: ({ signal }) =>
        api.posts(`?_page=${filters.page}&_limit=9`, signal),
      placeholderData: keepPreviousData,
      staleTime: 30_000,
    }),
  detail: (id) =>
    queryOptions({
      queryKey: postKeys.detail(id),
      queryFn: ({ signal }) => api.post(id, signal),
    }),
}

۳) Mutations در یک فایل

src/features/posts/mutations.jsjs
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { api } from '../../lib/api'
import { postKeys } from './keys'
import { useUIStore } from '../../stores/useUIStore'

export function useAddPost() {
  const qc = useQueryClient()
  const toast = useUIStore((s) => s.toast)

  return useMutation({
    mutationFn: (post) => api.addPost(post),
    onMutate: async (post) => {
      await qc.cancelQueries({ queryKey: postKeys.lists() })
      const previous = qc.getQueryData(postKeys.list({ q: '', page: 1 }))
      qc.setQueryData(postKeys.list({ q: '', page: 1 }), (old) =>
        old ? [{ id: Date.now(), ...post }, ...old] : old
      )
      return { previous }
    },
    onError: (_e, _v, ctx) => {
      qc.setQueryData(postKeys.list({ q: '', page: 1 }), ctx?.previous)
      toast('انتشار ناموفق بود')
    },
    onSuccess: () => toast('پست منتشر شد!'),
    onSettled: () => qc.invalidateQueries({ queryKey: postKeys.all }),
  })
}

export function useDeletePost() {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: (id) => api.deletePost(id),
    onMutate: async (id) => {
      await qc.cancelQueries({ queryKey: postKeys.lists() })
      const previous = qc.getQueryData(postKeys.list({ q: '', page: 1 }))
      qc.setQueryData(postKeys.list({ q: '', page: 1 }), (old) =>
        old?.filter((p) => p.id !== id) ?? old
      )
      return { previous }
    },
    onError: (_e, _v, ctx) =>
      qc.setQueryData(postKeys.list({ q: '', page: 1 }), ctx?.previous),
    onSettled: () => qc.invalidateQueries({ queryKey: postKeys.all }),
  })
}

۴) کامپوننت‌ها

src/components/PostList.jsxjsx
import { useMemo } from 'react'
import { useQuery } from '@tanstack/react-query'
import { postQueries } from '../features/posts/queries'
import { usePostFiltersStore } from '../stores/usePostFiltersStore'
import { PostCard } from './PostCard'

export default function PostList() {
  const q = usePostFiltersStore((s) => s.q)
  const page = usePostFiltersStore((s) => s.page)
  const setQ = usePostFiltersStore((s) => s.setQ)
  const nextPage = usePostFiltersStore((s) => s.nextPage)
  const prevPage = usePostFiltersStore((s) => s.prevPage)

  // debounce سبک: کلید فقط با مقدار پایدار عوض شود
  const debouncedQ = useDebounce(q, 300)

  const { data, isPending, isError, isPlaceholderData } = useQuery(
    postQueries.list({ q: debouncedQ, page })
  )

  const filtered = useMemo(
    () => (data ?? []).filter((p) => p.title.toLowerCase().includes(debouncedQ.toLowerCase())),
    [data, debouncedQ]
  )

  if (isPending) return <p>در حال بارگذاری پست‌ها...</p>
  if (isError) return <p>خطا در دریافت؛ بعداً دوباره تلاش کنید.</p>

  return (
    <section>
      <input
        value={q}
        onChange={(e) => setQ(e.target.value)}
        placeholder="جستجو در عنوان‌ها..."
      />
      <div style={{ opacity: isPlaceholderData ? 0.6 : 1 }}>
        {filtered.map((p) => <PostCard key={p.id} post={p} />)}
      </div>
      <nav>
        <button onClick={prevPage}>قبلی</button>
        <button onClick={nextPage}>بعدی</button>
      </nav>
    </section>
  )
}
src/components/Header.jsx — ورود/خروج + اتصال به کشjsx
import { useState } from 'react'
import { useMutation, useQueryClient } from '@tanstack/react-query'
import { api } from '../lib/api'
import { useAuthStore } from '../stores/useAuthStore'

export default function Header() {
  const user = useAuthStore((s) => s.user)
  const setSession = useAuthStore((s) => s.setSession)
  const clearAuth = useAuthStore((s) => s.clear)
  const qc = useQueryClient()
  const [email, setEmail] = useState('')

  const login = useMutation({
    mutationFn: (mail) => api.login(mail),
    onSuccess: (session) => setSession(session),
  })

  const logout = () => {
    clearAuth()
    qc.clear() // داده‌های سشن‌دار از کش هم بروند
  }

  return user ? (
    <header>
      <span>{user.name}</span>
      <button onClick={logout}>خروج</button>
    </header>
  ) : (
    <header>
      <input value={email} onChange={(e) => setEmail(e.target.value)} placeholder="email" />
      <button disabled={login.isPending} onClick={() => login.mutate(email)}>ورود</button>
    </header>
  )
}
