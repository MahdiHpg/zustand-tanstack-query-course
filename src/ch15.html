چیت‌شیت Zustand

zustand-cheatsheet.jsjs
import { create } from 'zustand'
import { persist, devtools } from 'zustand/middleware'
import { immer } from 'zustand/middleware/immer'
import { useShallow } from 'zustand/react/shallow'
import { subscribeWithSelector } from 'zustand/middleware'

// ---- ساخت استور (ترتیب میان‌افزار: devtools ← persist ← immer) ----
const useStore = create(
  devtools(
    persist(
      immer((set, get) => ({
        // State
        count: 0,
        items: [],

        // اکشن sync — تابع آپدیتر
        inc: (by = 1) => set((s) => ({ count: s.count + by })),

        // اکشن با immer (جهش امن)
        addItem: (item) => set((s) => { s.items.push(item) }),

        // اکشن async
        load: async () => {
          set({ status: 'loading' })
          try {
            const data = await api()
            set({ data, status: 'success' })
          } catch (e) {
            set({ status: 'error', error: e.message })
          }
        },

        // خواندن آخرین State با get
        double: () => set({ count: get().count * 2 }),

        // ریست
        reset: () => set({ count: 0, items: [] }),
      })),
      { name: 'key', partialize: (s) => ({ count: s.count }) }
    )
  )
)

// ---- مصرف در کامپوننت ----
const count  = useStore((s) => s.count)          // سلکتور تکی (پیش‌فرض)
const { a, b } = useStore(useShallow((s) => ({ a: s.a, b: s.b }))) // چند فیلد

// ---- بیرون React ----
useStore.getState().inc()        // خواندن/اجرا بدون subscribe
useStore.setState({ count: 5 })  // نوشتن مستقیم
const unsub = useStore.subscribe((s) => console.log(s.count)) // گوش به همه
useStore.persist.rehydrate()     // بازیابی دستی (SSR)

// ---- TS: فرم کِرید ----
// create<State>()((set) => ({ ... }))

چیت‌شیت TanStack Query

tanstack-query-cheatsheet.jsxjsx
import { QueryClient, QueryClientProvider, useQuery, useMutation,
         useQueryClient, keepPreviousData, useInfiniteQuery,
         useSuspenseQuery } from '@tanstack/react-query'

// ---- راه‌اندازی (یک‌بار) ----
const qc = new QueryClient({
  defaultOptions: { queries: { staleTime: 60_000, retry: 2 } },
})
<QueryClientProvider client={qc}><App /></QueryClientProvider>

// ---- Query Key Factory ----
const postKeys = {
  all: ['posts'],
  lists: () => [...postKeys.all, 'list'],
  list: (f) => [...postKeys.lists(), f],
  detail: (id) => [...postKeys.all, 'detail', id],
}

// ---- خواندن ----
const { data, isPending, isError, error, isFetching, isPlaceholderData } =
  useQuery({
    queryKey: postKeys.list({ page }),
    queryFn: ({ signal }) => api.posts(page, signal),
    placeholderData: keepPreviousData, // صفحه‌بندی نرم
    enabled: !!userId,                 // کوئری وابسته
    select: (d) => d.filter(Boolean),  // تبدیل
    staleTime: 30_000,
  })

// اسکرول بی‌نهایت
useInfiniteQuery({
  queryKey: postKeys.lists(),
  queryFn: ({ pageParam, signal }) => api.posts(pageParam, signal),
  initialPageParam: 1,
  getNextPageParam: (last, all, lastParam) => last.next ?? undefined,
})

// ---- تغییر + invalidation ----
const mutation = useMutation({
  mutationFn: (vars) => api.addPost(vars),
  onSuccess: () => qc.invalidateQueries({ queryKey: postKeys.lists() }),
})
mutation.mutate({ title })

// ---- Optimistic Update (حفظ کنید!) ----
onMutate: async (v) => {
  await qc.cancelQueries({ queryKey: postKeys.lists() })
  const prev = qc.getQueryData(postKeys.list(f))
  qc.setQueryData(postKeys.list(f), (old) => update(old, v))
  return { prev }
},
onError: (_e, _v, ctx) => qc.setQueryData(postKeys.list(f), ctx.prev),
onSettled: () => qc.invalidateQueries({ queryKey: postKeys.lists() }),

// ---- ابزارهای دستوری ----
qc.setQueryData(key, updater)     // نوشتن مستقیم در کش
qc.getQueryData(key)              // خواندن کش
qc.prefetchQuery({ queryKey, queryFn })   // پیش‌واکشی (خطاها بی‌صدا)
qc.ensureQueryData({ queryKey })  // گرفتن مطمئن (لودرها)
qc.removeQueries({ queryKey })    // حذف از کش
qc.clear()                        // خروج کاربر

اصطلاحمعادل و توضیح فارسی
Stateوضعیت — دادهٔ تغییرپذیری که UI از آن رندر می‌شود

Middlewareمیان‌افزار — لایهٔ ارتقادهندهٔ رفتار استور (persist و...)

Hydration / Rehydrateآب‌رسانی — بازیابی دادهٔ ذخیره‌شده به استور پس از لود
Immutabilityتغییرناپذیری — ساخت نسخهٔ جدید به‌جای جهش دادهٔ موجود
Shallow Compareمقایسهٔ سطحی — برابری کلید‌ها/مقادیر یک سطح

Client Stateوضعیت کلاینت — داده‌ای که مرورگر مالک حقیقی آن است
