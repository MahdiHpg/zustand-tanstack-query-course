پروژهٔ عملی ۱ — سبد خرید کامل با Zustand

همهٔ آموخته‌های پنج فصل را در یک پروژهٔ واقعی جمع می‌کنیم: فروشگاه کوچکی با گرید محصولات، سبد خرید کشویی، کد تخفیف، محاسبات مشتق‌شده و persist.

در این فصل یاد می‌گیرید

  • طراحی استور برای یک دامنهٔ واقعی قبل از نوشتن حتی یک خط UI
  • جداسازی State داده از State UI
  • محاسبات مشتق‌شده (جمع، تخفیف، نهایی) بدون تکرار داده
  • اتصال persist و مدیریت hydration در یک پروژهٔ کامل

طراحی قبل از کد

درخواست محصول: «کاربر محصولات را می‌بیند، به سبد اضافه می‌کند، تعداد را تغییر می‌دهد، کد تخفیف می‌زند و جمع نهایی را می‌بیند. سبد بعد از رفرش باقی بماند.» قبل از هر کدی، State را طراحی می‌کنیم:

۲) استور سبد — قلب پروژه

src/store/useCartStore.jsjs
import { create } from 'zustand'
import { persist } from 'zustand/middleware'
import { COUPONS } from '../data/products'

const initialState = { items: [], couponCode: null, _hasHydrated: false }

export const useCartStore = create(
  persist(
    (set, get) => ({
      ...initialState,

      addItem: (id) =>
        set((s) => {
          const found = s.items.find((i) => i.id === id)
          if (found) {
            return { items: s.items.map((i) => (i.id === id ? { ...i, qty: i.qty + 1 } : i)) }
          }
          return { items: [...s.items, { id, qty: 1 }] }
        }),

      setQty: (id, qty) =>
        set((s) => ({
          items:
            qty <= 0
              ? s.items.filter((i) => i.id !== id)
              : s.items.map((i) => (i.id === id ? { ...i, qty } : i)),
        })),

      removeItem: (id) => set((s) => ({ items: s.items.filter((i) => i.id !== id) })),

      applyCoupon: (code) => {
        const coupon = COUPONS[code.trim().toUpperCase()]
        if (!coupon) return { ok: false, message: 'کد تخفیف نامعتبر است' }
        const subtotal = get().items.reduce((sum, i) => sum + i.qty * priceOf(i.id), 0)
        if (subtotal < coupon.min) {
          return { ok: false, message: 'حداقل مبلغ برای این کد برآورده نشده' }
        }
        set({ couponCode: code.trim().toUpperCase() })
        return { ok: true, message: 'کد تخفیف اعمال شد' }
      },

      removeCoupon: () => set({ couponCode: null }),
      clearCart: () => set({ items: [], couponCode: null }),
      setHasHydrated: (v) => set({ _hasHydrated: v }),
    }),
    {
      name: 'shop-cart',
      partialize: (s) => ({ items: s.items, couponCode: s.couponCode }),
      onRehydrateStorage: () => (state) => state?.setHasHydrated(true),
    }
  )
)

// کمکی: قیمت محصول از کاتالوگ (منبع واحد حقیقت)
import { PRODUCTS } from '../data/products'
const priceOf = (id) => PRODUCTS.find((p) => p.id === id)?.price ?? 0

به applyCoupon دقت کنید: از get() برای خواندن جمع فعلی استفاده کرد و «نتیجهٔ اعتبارسنجی» را به‌صورت مقدار برگرداند — کامپوننت می‌تواند بر اساس آن پیام نشان دهد؛ به‌جای اینکه منطق پیام‌ها داخل استور بنشیند.

src/components/CartDrawer.jsxjsx
import { useMemo, useState } from 'react'
import { PRODUCTS } from '../data/products'
import { useCartStore } from '../store/useCartStore'
import { useShopUIStore } from '../store/useShopUIStore'

const byId = Object.fromEntries(PRODUCTS.map((p) => [p.id, p]))

export default function CartDrawer() {
  const cartOpen = useShopUIStore((s) => s.cartOpen)
  const closeCart = useShopUIStore((s) => s.closeCart)
  const items = useCartStore((s) => s.items)
  const couponCode = useCartStore((s) => s.couponCode)
  const setQty = useCartStore((s) => s.setQty)
  const removeItem = useCartStore((s) => s.removeItem)
  const clearCart = useCartStore((s) => s.clearCart)
  const applyCoupon = useCartStore((s) => s.applyCoupon)
  const removeCoupon = useCartStore((s) => s.removeCoupon)
  const hasHydrated = useCartStore((s) => s._hasHydrated)

  const [code, setCode] = useState('')
  const [msg, setMsg] = useState(null)

  // محاسبات مشتق‌شده — فقط وقتی items یا couponCode عوض شود
  const totals = useMemo(() => {
    const subtotal = items.reduce((sum, i) => sum + byId[i.id].price * i.qty, 0)
    let discount = 0
    if (couponCode === 'YALDA10') discount = Math.round(subtotal * 0.1)
    if (couponCode === 'SHIP500') discount = 500_000
    return { subtotal, discount, payable: subtotal - discount }
  }, [items, couponCode])

  if (!cartOpen) return null

  return (
    <aside className="drawer">
      <header>
        <h2>سبد خرید ({items.length.toLocaleString('fa-IR')})</h2>
        <button onClick={closeCart}>بستن</button>
      </header>

      {!hasHydrated && <p>در حال بازیابی سبد...</p>}

      {items.map((i) => (
        <div key={i.id} className="row">
          <span>{byId[i.id].emoji} {byId[i.id].name}</span>
          <span>
            <button onClick={() => setQty(i.id, i.qty - 1)}>−</button>
            {' '}{i.qty.toLocaleString('fa-IR')}{' '}
            <button onClick={() => setQty(i.id, i.qty + 1)}>+</button>
          </span>
          <button onClick={() => removeItem(i.id)}>حذف</button>
        </div>
      ))}

      {msg && <p className={msg.ok ? 'ok' : 'err'}>{msg.message}</p>}

      {couponCode ? (
        <button onClick={removeCoupon}>حذف کد {couponCode}</button>
      ) : (
        <div className="coupon">
          <input value={code} onChange={(e) => setCode(e.target.value)} placeholder="کد تخفیف" />
          <button onClick={() => setMsg(applyCoupon(code))}>اعمال</button>
        </div>
      )}

      <footer>
        <div>جمع: {totals.subtotal.toLocaleString('fa-IR')}</div>
        <div>تخفیف: {totals.discount.toLocaleString('fa-IR')}</div>
        <div>قابل پرداخت: <b>{totals.payable.toLocaleString('fa-IR')}</b></div>
        <button disabled={items.length === 0} onClick={clearCart}>پرداخت و خالی کردن سبد</button>
      </footer>
    </aside>
  )
}

نقاط آموزشی پروژه

  • دو استور، دو دنیا: useCartStore داده است و useShopUIStore فقط نمایش. کامپوننت گرید محصول از دومی فقط «باز کردن» می‌گیرد؛ از محتوای سبد خبر ندارد.
  • سلکتور تکی همه‌جا: هر خط useXStore((s) => ...) یک فیلد است — دقیقاً همان که فصل ۴ گفت.
  • مرجع پایدار برای useMemo: چون items فقط هنگام mutation جدید می‌شود، totals کم محاسبه می‌شود.
  • hydration فلگ: پیام «در حال بازیابی» تا تمام شدن rehydrate رندر می‌شود و بعد لیست واقعی.
هشدار — مرحلهٔ بعدی همان تلهٔ کلاسیک است

اگر همین پروژه بخواهد «موجودی واقعی» را از سرور بگیرد، اولین وسوسه این است که موجودی را در useCartStore ذخیره کنیم! از فصل ۸ می‌بینیم چرا دادهٔ سرور باید در کشِ TanStack Query برود و Zustand فقط «سبدِ خود کاربر» را نگه دارد.

جمع‌بندی فصل

  • اول State را روی کاغذ طراحی کنید؛ بعد UI را بنویسید.
  • در سبد فقط id و qty نگه دارید؛ جزئیات را از منبع اصلی بخوانید.
  • محاسبات مشتق‌شده ذخیره نمی‌شوند؛ با useMemo در کامپوننت حساب می‌شوند.
  • اعتبارسنجی (کد تخفیف) داخل اکشن با get() و برگرداندن نتیجه به کامپوننت.
  • persist با partialize + فلگ hydration برای تجربهٔ بدون فلش.
تمرین فصل ۷ — سه ارتقا

۱) قابلیت «ذخیره برای بعد» اضافه کنید: لیست علاقه‌مندی‌ها (wishlist) در استور جدا با persist مشترک؟ نه — استور مستقل با نام کلید خودش. ۲) دکمهٔ «خالی کردن» با تأیید دو مرحله‌ای (State confirming در UI استور). ۳) کد تخفیف جدید بنویسید که اگر تعداد کل آیتم‌ها از ۵ بیشتر بود ۱۵٪ بدهد — سرنخ: در applyCoupon به items.length نیاز دارید.
